    <script>
        function formatRupiah(angka) {
            return new Intl.NumberFormat('id-ID', { style: 'currency', currency: 'IDR', maximumFractionDigits: 0 }).format(angka);
        }

        function hitungGaji() {
            const paket = parseFloat(document.getElementById('paket').value) || 0;
            const rate = parseFloat(document.getElementById('rate').value) || 0;
            const hari = parseFloat(document.getElementById('hari').value) || 0;
            const operasional = parseFloat(document.getElementById('operasional').value) || 0;

            const kotorPerHari = paket * rate;
            const bersihPerHari = kotorPerHari - operasional;
            const kotorPerBulan = kotorPerHari * hari;
            const bersihPerBulan = bersihPerHari * hari;

            document.getElementById('kotorHari').innerText = formatRupiah(kotorPerHari);
            document.getElementById('bersihHari').innerText = formatRupiah(bersihPerHari);
            document.getElementById('kotorBulan').innerText = formatRupiah(kotorPerBulan);
            document.getElementById('bersihBulan').innerText = formatRupiah(bersihPerBulan);

            document.getElementById('resultBox').style.display = 'block';
        }
    </script>
